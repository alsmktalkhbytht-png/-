def _geno(a, b, order):
    pair = sorted([a, b], key=lambda x: order[x])
    return pair[0] + pair[1]

def PS(g1, g2, order, p1=("", ""), p2=("", "")):
    """Punnett square: rows = gametes of parent 1, cols = gametes of parent 2."""
    head = [B("Gametes", "الأمشاج")] + [f"`{g}`" for g in g2]
    rows = [[f"`{a}`"] + [f"`{_geno(a, b, order)}`" for b in g2] for a in g1]
    w = 100 / (len(g2) + 1.4)
    return T(head, rows, key_cols=[0], widths=[round(w * 1.4)] + [round(w)] * len(g2))
